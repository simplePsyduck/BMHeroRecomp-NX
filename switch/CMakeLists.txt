# Nintendo Switch build of Bomberman Hero (BMHero_nx). Own root, like DK64Recomp-NX's switch/ directory, so the
# upstream CMakeLists (RT64, RmlUi, X11, Windows/macOS bundles) stays untouched. Build with switch/build.sh.
#
# Structure borrowed from DK64Recomp-NX (GPL-3.0). Nothing here includes game assets: RecompiledFuncs/ and
# RecompiledPatches/ are generated locally from your own ROM by switch/codegen.sh and are git-ignored.
cmake_minimum_required(VERSION 3.24)
project(BMHero_nx LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF)

# Renderer: "gliden64" (Phase 1, default) or "deko3d" (Phase 2, needs switch/render_deko3d.cpp).
set(BMHNX_RENDERER "gliden64" CACHE STRING "Renderer backend: gliden64 or deko3d")
set_property(CACHE BMHNX_RENDERER PROPERTY STRINGS gliden64 deko3d)

set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

# Shim headers (sys/mman.h, dlfcn.h, recompui/, recompinput/) take precedence everywhere, including inside
# N64ModernRuntime.
include_directories(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/shim)
add_compile_definitions(_GNU_SOURCE=1 FMT_USE_FALLBACK_FILE=1)
add_compile_options(-fno-omit-frame-pointer)
get_filename_component(ROOT_ABS ${ROOT} ABSOLUTE)
add_compile_options(-ffile-prefix-map=${ROOT_ABS}/=)
set(FMT_OS OFF CACHE BOOL "" FORCE)

add_subdirectory(${ROOT}/lib/N64ModernRuntime ${CMAKE_BINARY_DIR}/N64ModernRuntime)
# librecomp's mods.cpp (patched) writes code through bmhnx_write_code, and LiveRecomp compiles mod code with sljit
# into a libnx Jit (sljit/sljitConfigPre.h, jit_switch.c). Mods are not loaded in Phase 1; this only keeps it linking.
target_compile_definitions(LiveRecomp PRIVATE SLJIT_HAVE_CONFIG_PRE=1)
target_include_directories(LiveRecomp BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/sljit)

find_package(SDL2 REQUIRED)

set(RECOMP_INCLUDES
    ${ROOT}/include
    ${ROOT}/lib/N64ModernRuntime/ultramodern/include
    ${ROOT}/lib/N64ModernRuntime/librecomp/include
    ${ROOT}/lib/N64ModernRuntime/N64Recomp/include
)
set(RECOMP_C_FLAGS
    -fno-strict-aliasing
    -falign-functions=16
    -Wno-unused-variable
    $<$<COMPILE_LANGUAGE:C>:-Wno-implicit-function-declaration>
    $<$<COMPILE_LANGUAGE:C>:-Wno-int-conversion>
    $<$<COMPILE_LANGUAGE:C>:-Wno-incompatible-pointer-types>
)

# Generated by switch/codegen.sh (N64Recomp bmhero.toml).
file(GLOB FUNC_SOURCES ${ROOT}/RecompiledFuncs/*.c ${ROOT}/RecompiledFuncs/*.cpp)
if(NOT FUNC_SOURCES)
    message(FATAL_ERROR "RecompiledFuncs/ is empty: run switch/codegen.sh with your own ROM first")
endif()
add_library(RecompiledFuncs STATIC ${FUNC_SOURCES})
target_include_directories(RecompiledFuncs PRIVATE ${RECOMP_INCLUDES})
target_compile_options(RecompiledFuncs PRIVATE ${RECOMP_C_FLAGS})
# GCC: originals weak, patches strong (see N64Recomp/include/recomp.h, patched by switch/patches/*.patch).
target_compile_definitions(RecompiledFuncs PRIVATE RECOMP_FUNC_WEAK)

# The patch blob is embedded as bk_patches_bin (what src/main/register_patches.cpp declares), generated here from
# patches/patches.bin so no host tools are needed. The output lives in the build directory.
file(READ ${ROOT}/patches/patches.bin PATCHES_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PATCHES_BYTES "${PATCHES_HEX}")
file(WRITE ${CMAKE_BINARY_DIR}/patches_bin.c
    "#include <stddef.h>\n"
    "extern const char bk_patches_bin[];\n"
    "const char bk_patches_bin[] = {${PATCHES_BYTES}};\n"
    "extern const size_t bk_patches_bin_size;\n"
    "const size_t bk_patches_bin_size = sizeof(bk_patches_bin);\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ROOT}/patches/patches.bin)

add_library(PatchesLib STATIC
    ${ROOT}/RecompiledPatches/patches.c
    ${CMAKE_BINARY_DIR}/patches_bin.c
)
set_source_files_properties(${CMAKE_BINARY_DIR}/patches_bin.c PROPERTIES COMPILE_FLAGS -Wno-overflow)
target_include_directories(PatchesLib PRIVATE ${RECOMP_INCLUDES})
target_compile_options(PatchesLib PRIVATE ${RECOMP_C_FLAGS})

# ---- Renderer: GLideN64 on OpenGL through mesa/nouveau (as DK64Recomp-NX).
set(GLIDEN64_DIR ${ROOT}/lib/GLideN64/src)
include(gliden64_sources.cmake)
list(REMOVE_ITEM GLIDEN64_SOURCES ${GLIDEN64_DIR}/CRC32.cpp ${GLIDEN64_DIR}/3DMath.cpp ${GLIDEN64_DIR}/RSP_LoadMatrix.cpp)
list(APPEND GLIDEN64_SOURCES
    ${GLIDEN64_DIR}/CRC32_ARMV8.cpp
    ${GLIDEN64_DIR}/Neon/3DMathNeon.cpp
    ${GLIDEN64_DIR}/Neon/gSPNeon.cpp
    ${GLIDEN64_DIR}/Neon/RSP_LoadMatrixNeon.cpp
)
add_library(GLideN64 STATIC ${GLIDEN64_SOURCES} gliden64_rt64ex.cpp)
target_include_directories(GLideN64 PRIVATE
    ${GLIDEN64_DIR} ${GLIDEN64_DIR}/inc ${GLIDEN64_DIR}/osal
    ${CMAKE_CURRENT_SOURCE_DIR}/gliden64
    $ENV{DEVKITPRO}/portlibs/switch/include
)
target_compile_definitions(GLideN64 PRIVATE MUPENPLUSAPI TXFILTER_LIB OS_LINUX EGL NDEBUG __NEON_OPT __VEC4_OPT)
set_target_properties(GLideN64 PROPERTIES CXX_STANDARD 17)
target_compile_options(GLideN64 PRIVATE -Wno-narrowing $<$<COMPILE_LANGUAGE:CXX>:-fpermissive>)

# Pause menu (nx_menu.cpp): Dear ImGui at the version RT64 pins, with its OpenGL 3 backend; glad (a devkitPro portlib)
# loads the GL functions.
set(IMGUI_DIR ${ROOT}/lib/rt64/src/contrib/imgui)
add_library(imgui STATIC
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
)
target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends $ENV{DEVKITPRO}/portlibs/switch/include)
target_compile_definitions(imgui PRIVATE IMGUI_IMPL_OPENGL_LOADER_CUSTOM)
set_source_files_properties(${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp PROPERTIES COMPILE_OPTIONS "-include;glad/glad.h")
target_link_libraries(imgui PUBLIC glad)

add_executable(BMHero_nx
    ${ROOT}/src/game/debug.cpp
    ${ROOT}/src/game/recomp_api.cpp
    ${ROOT}/src/game/recomp_data_api.cpp
    ${ROOT}/src/game/recomp_extension_api.cpp
    ${ROOT}/src/game/rom_decompression.cpp
    ${ROOT}/src/main/register_overlays.cpp
    ${ROOT}/src/main/register_patches.cpp
    ${ROOT}/rsp/n_aspMain.cpp
    core_switch.cpp
    frontend_switch.cpp
    input_switch.cpp
    main_switch.cpp
    mman_switch.c
    threads_switch.c
    crash_switch.c
    render_gliden64.cpp
    renderer_select.cpp
    nx_config.cpp
    nx_controls.cpp
    nx_settings.cpp
    nx_menu.cpp
    nx_mods.cpp
    nx_menu_widgets.cpp
    vsync_switch.cpp
    window_switch.c
    jit_switch.c
    perf_switch.cpp
)
if(BMHNX_RENDERER STREQUAL "deko3d")
    if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/render_deko3d.cpp)
        message(FATAL_ERROR "BMHNX_RENDERER=deko3d needs switch/render_deko3d.cpp (Phase 2)")
    endif()
    target_sources(BMHero_nx PRIVATE render_deko3d.cpp)
    target_compile_definitions(BMHero_nx PRIVATE BMHNX_HAVE_DEKO3D BMHNX_DEFAULT_DEKO3D)
    target_link_libraries(BMHero_nx PRIVATE deko3d)
endif()
target_include_directories(BMHero_nx PRIVATE
    ${RECOMP_INCLUDES}
    ${ROOT}/lib/N64ModernRuntime/thirdparty/sse2neon
    ${ROOT}/lib/SlotMap
    ${ROOT}/lib/GLideN64/src/inc
    # banjo_config.h includes "json/json.hpp"; BMHeroRecomp's own target finds it through RT64's contrib dir.
    ${ROOT}/lib/rt64/src/contrib
)
target_compile_options(BMHero_nx PRIVATE -fno-strict-aliasing -fms-extensions)
# libnx gives std::thread 128 KiB stacks and cannot detach threads: see threads_switch.c.
target_link_options(BMHero_nx PRIVATE -Wl,--wrap=pthread_create,--wrap=pthread_detach)
# The recompiled libraries call into each other in both directions.
set(CMAKE_CXX_LINK_GROUP_USING_RESCAN "LINKER:--start-group" "LINKER:--end-group")
set(CMAKE_CXX_LINK_GROUP_USING_RESCAN_SUPPORTED TRUE)
target_link_libraries(BMHero_nx PRIVATE
    "$<LINK_GROUP:RESCAN,PatchesLib,RecompiledFuncs,librecomp,ultramodern>"
    GLideN64
    imgui
    SDL2::SDL2
    EGL
    GLESv2
    glapi
    drm_nouveau
    nx
)

# What the launcher shows is "Bomberman Hero" (BMHNX_DISPLAY_NAME in nx_paths.h); the file is BMHero_nx.nro.
nx_generate_nacp(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/BMHero_nx.nacp
    NAME "Bomberman Hero"
    AUTHOR "simplePsyduck"
    VERSION "0.7.3"
)
nx_create_nro(BMHero_nx_nro
    TARGET BMHero_nx
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/BMHero_nx.nro
    ICON ${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg
    NACP ${CMAKE_CURRENT_BINARY_DIR}/BMHero_nx.nacp
)
